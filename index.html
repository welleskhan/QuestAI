async function chamarAPIProviaProxy(provider, model, prompt) { try { const response = await fetch('/api/ai-proxy', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ provider, model, prompt }) }); const data = await response.json(); if (!response.ok) throw new Error(data.error || 'Erro na API interna'); return data.result; } catch (err) { throw new Error(`Falha na comunicação: ${err.message}`); } }
